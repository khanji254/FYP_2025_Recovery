\begin{frame}{Introduction \& Background}
\begin{columns}[T]
  \begin{column}{0.54\textwidth}
    \textbf{Context:}
    \begin{itemize}
      \item The Nakuja Project is JKUAT's student rocketry programme, targeting altitude records with successive N-series rockets.
      \item The N4 rocket uses a \textbf{dual-deployment recovery system} â€” a drogue parachute at apogee stabilises descent; a main parachute deploys at low altitude for safe landing.
      \item Recovery depends on three tightly coupled systems: \alert{telemetry} (track the rocket after separation), \alert{pyrotechnic deployment} (eject the parachutes reliably), and \alert{pre-flight monitoring} (confirm readiness before launch).
      \item The N3 predecessor exposed reliability gaps in all three areas.
    \end{itemize}

% SLIDE 5 â€” OBJECTIVES
% ==================================================================
\begin{frame}{Objectives}
\begin{block}{Main Objective}
To design, fabricate, and validate a reliable dual-deployment recovery system for
the N4 rocket, achieving telemetry coverage beyond 5~km, controlled pyrotechnic
deployment, and real-time pre-flight system diagnostics.
\end{block}
\vspace{0.5em}
\textbf{Specific Objectives:}
\begin{enumerate}
  \item \textbf{Telemetry range:} Design and configure a 900~MHz XBee-PRO 900HP radio link
        with a verified link margin $\geq$20~dB at 5~km range to ensure post-apogee
        rocket tracking and telemetry reception.
  \item \textbf{Controlled deployment:} Implement a PWM-based ignition controller driving
        an IRLZ44N MOSFET at a calibrated duty cycle to eliminate nichrome burnout and
        provide safe, repeatable pyrotechnic ejection charge activation.
  \item \textbf{Structural design \& in-flight monitoring:} Design a 3D-printed PETG
        avionics holder satisfying the 95~mm bay envelope constraint; integrate an
        ADS1115 16-bit ADC circuit for battery voltage and igniter continuity monitoring
        with launch-inhibit logic.
\end{enumerate}
\end{frame}

% SLIDE 8 â€” GAPS & PROPOSED APPROACH
% ==================================================================
\begin{frame}{Gaps Identified \& Proposed Approach}
\begin{columns}[T]
  \begin{column}{0.50\textwidth}
    \textbf{Gaps in existing solutions:}
    \begin{itemize}
      \item No verified 5~km link budget for N4 attitude uncertainty â€” literature gives theory only; no field-proven margin for this specific flight profile.
      \item DC pyrotechnic dump is single-use and thermally destructive â€” no study combines PWM ignition with a logic-level MOSFET on an ESP32 platform.
      \item No integrated launch-inhibit system combining battery voltage monitoring \emph{and} continuity detection on a single flight computer.
      \item No compact PETG avionics sled designed for the specific 95~mm N4 bay with threaded-rod battery retention.
    \end{itemize}
  \end{column}
  \begin{column}{0.47\textwidth}
    \textbf{Proposed approach:}
    \begin{enumerate}\setlength{\itemsep}{2pt}
      \item \alert{XBee-PRO 900HP} â€” 900~MHz, +24~dBm; calculated link margin +32~dB at 5~km.
      \item \alert{PWM ignition} at 1~kHz, $D=0.40$; IRLZ44N MOSFET driven directly from 3.3~V ESP32 GPIO.
      \item \alert{ADS1115 I$^2$C ADC} â€” 47~k$\Omega$/10~k$\Omega$ divider; launch inhibit if $V_{bat}<14.0$~V or continuity absent.
      \item \alert{Stepped PETG battery carrier} â€” raised edge cells clear threaded rods; dual-rod retention for anti-vibration.
      \item \alert{FreeRTOS task firmware} â€” modular, independently testable tasks; FSM state machine.
    \end{enumerate}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Methodology â€” Pyrotechnic Deployment Module}
\begin{columns}[T]

  \begin{column}{0.50\textwidth}
    \textbf{Design Considerations:}
    \begin{itemize}
      \item Must fire reliably, but \emph{not} burn out the igniter
      \item Logic-level control from 3.3~V ESP32 GPIO
      \item Inductive flyback protection for solenoid-type loads
    \end{itemize}
    \vspace{0.3em}
    \textbf{DC dump vs PWM trade study:}
    \begin{tabular}{@{}p{2.2cm}p{1.8cm}p{1.8cm}@{}}
      \toprule
      \textbf{Criterion} & \textbf{DC Dump} & \textbf{PWM}\\
      \midrule
      Wire re-use    & No          & Yes \\
      Current control & None       & $\checkmark$ \\
      Holder damage  & High risk   & Mitigated \\
      Reliability    & Single-use  & Repeatable \\
      \bottomrule
    \end{tabular}
    \vspace{0.3em}
    \textbf{Selected:} PWM at 1~kHz, $D=0.40$ ($V_{eff}\approx6$~V from 15~V bus)\\
    Driver: IRLZ44N MOSFET ($V_{GS(th)}<2$~V; $R_{DS(on)}=22$~m$\Omega$)
  \end{column}
