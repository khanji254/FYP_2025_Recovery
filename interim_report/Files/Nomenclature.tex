\section*{Nomenclature}
\addcontentsline{toc}{section}{Nomenclature}

\textbf{Abbreviations}
\begin{description}
    \item[ADC] Analog-to-Digital Converter
    \item[AGL] Above Ground Level
    \item[BMS] Battery Management System
    \item[CAD] Computer-Aided Design
    \item[ESP32] Espressif 32-bit Microcontroller Platform
    \item[FEA] Finite Element Analysis
    \item[GPS] Global Positioning System
    \item[IMU] Inertial Measurement Unit
    \item[LiPo] Lithium Polymer Battery
    \item[MOSFET] Metal-Oxide-Semiconductor Field-Effect Transistor
    \item[PWM] Pulse Width Modulation
    \item[RF] Radio Frequency
    \item[UART] Universal Asynchronous Receiver-Transmitter
\end{description}

\textbf{Symbols and Units}
\begin{description}
    \item[$D$] Duty cycle, percent
    \item[$f$] Frequency, hertz
    \item[$I$] Current, ampere
    \item[$P$] Power, watt
    \item[$R$] Resistance, ohm
    \item[$T$] Temperature, degree Celsius
    \item[$V$] Voltage, volt
\end{description}
