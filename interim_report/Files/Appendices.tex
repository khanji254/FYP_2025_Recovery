\section{Appendix A: Component Specifications}
\label{app:components}

\subsection{ESP32 Microcontroller}
\begin{itemize}
    \item Model: ESP32-WROOM-32
    \item CPU: Dual-core Xtensa LX6, 240 MHz
    \item Memory: 520 KB SRAM, 4 MB Flash
    \item Interfaces: 3× UART, 2× I$^2$C, 3× SPI, 18× ADC (12-bit)
    \item Power: 3.3V operation, 80--260 mA active current
    \item Operating Temperature: -40 to +85°C
\end{itemize}

\subsection{XBee-PRO 900HP}
\begin{itemize}
    \item Frequency: 902--928 MHz (ISM band)
    \item RF Power: Up to 250 mW (+24 dBm)
    \item Receiver Sensitivity: -110 dBm (at 10 kbps)
    \item Range: Up to 15 km line-of-sight
    \item Data Rate: 10 kbps to 200 kbps (configurable)
    \item Interface: UART (3.3V logic)
    \item Power: 3.3V, 215 mA TX, 35 mA RX
\end{itemize}

\subsection{BMS Board (4S LiPo)}
\begin{itemize}
    \item Configuration: 4S (14.8V nominal)
    \item Charge Current: Up to 15A
    \item Discharge Current: 30A continuous, 40A peak
    \item Over-voltage Protection: 4.25V per cell
    \item Under-voltage Protection: 3.0V per cell
    \item Balance Tolerance: ±0.05V
\end{itemize}

\subsection{IRLZ44N MOSFET}
\begin{itemize}
    \item Type: N-channel enhancement mode
    \item Voltage Rating: 55V
    \item Current Rating: 47A continuous
    \item On-Resistance: 22 m$\Omega$ @ 10V $V_{GS}$
    \item Gate Threshold: 1--2V (logic-level)
    \item Package: TO-220AB
\end{itemize}

\subsection{Nichrome Wire (Igniter)}
\begin{itemize}
    \item Type: Nichrome A (80\% Ni, 20\% Cr)
    \item Gauge: 34 AWG (0.16 mm diameter)
    \item Resistance: 34.5 $\Omega$/meter
    \item Operating Temperature: Up to 1150°C
    \item Wire Length per Igniter: 0.15--0.20 m
    \item Total Resistance per Igniter: 5--7 $\Omega$
\end{itemize}

\section{Appendix B: PCB Design Details}
\label{app:pcb}

\subsection{PCB Specifications}
\begin{itemize}
    \item Dimensions: 65 mm × 65 mm
    \item Layers: 2 (top and bottom copper)
    \item Copper Weight: 2 oz (70 $\mu$m)
    \item Material: FR-4, 1.6 mm thickness
    \item Surface Finish: ENIG (Electroless Nickel Immersion Gold)
    \item Via Size: 0.4 mm drill, 0.8 mm pad
    \item Trace Width: 0.25 mm minimum, 1.0 mm for high-current paths
\end{itemize}

\subsection{Power Distribution}
\begin{itemize}
    \item Input: 14.8V from 4S LiPo via XT60 connector
    \item 5V Rail: LM2596 buck converter, 3A capacity
    \item 3.3V Rail: AMS1117-3.3 LDO, 1A capacity
    \item Power Planes: Dedicated ground plane on bottom layer
    \item Decoupling: 100 nF ceramic capacitors on all ICs, 100 $\mu$F electrolytic on power inputs
\end{itemize}

\subsection{Circuit Protection}
\begin{itemize}
    \item Input Protection: Reverse polarity protection diode (Schottky, 40V, 5A)
    \item Over-current: Self-resetting fuses (PTC) on 5V and 3.3V rails
    \item MOSFET Protection: Flyback diodes on inductive loads
    \item ESD Protection: TVS diodes on communication lines
\end{itemize}

\section{Appendix E: Software Architecture}
\label{app:software}

\subsection{FreeRTOS Task Structure}

\begin{enumerate}
    \item \textbf{Sensor Task (10 Hz):} Reads GPS, IMU, barometer. Updates shared state variables via mutex.
    
    \item \textbf{Telemetry Task (5 Hz):} Formats telemetry packet, transmits via XBee UART, receives ground commands.
    
    \item \textbf{State Machine Task (20 Hz):} Manages flight phases, triggers deployment events, implements safety checks.
    
    \item \textbf{Deployment Task (Event-driven):} Executes deployment sequence when triggered by state machine. Activates PWM, monitors current.
    
    \item \textbf{Ignition Task (50 Hz):} Real-time PWM control with closed-loop current monitoring and thermal protection.
    
    \item \textbf{Logging Task (10 Hz):} Writes sensor data and events to SD card in CSV format.
    
    \item \textbf{Monitoring Task (1 Hz):} Checks battery voltage, igniter continuity, telemetry link health. Sets launch inhibit flags.
\end{enumerate}

\subsection{Deployment Sequence Flowchart}

\begin{enumerate}
    \item \textbf{Trigger Condition Detected:} Apogee or main altitude threshold
    \item \textbf{Check Safety Conditions:} Minimum altitude, minimum time, no previous deployment
    \item \textbf{Safety Checks Passed?}
    \begin{itemize}
        \item No: Abort, log error
        \item Yes: Continue
    \end{itemize}
    \item \textbf{Check Igniter Continuity:} Read continuity status
    \item \textbf{Continuity Good?}
    \begin{itemize}
        \item No: Abort, log error, trigger backup timer
        \item Yes: Continue
    \end{itemize}
    \item \textbf{Activate PWM:} Start at 10\% duty cycle
    \item \textbf{Ramp PWM:} Linear ramp to 70\% over 2 seconds
    \item \textbf{Hold PWM:} Maintain 70\% for 3 seconds
    \item \textbf{Monitor Current:} Verify expected current draw
    \item \textbf{Current Normal?}
    \begin{itemize}
        \item No: Log warning, continue
        \item Yes: Continue
    \end{itemize}
    \item \textbf{Disable PWM:} Turn off after ignition window
    \item \textbf{Mark Deployment Complete:} Set deployment flag, prevent re-trigger
    \item \textbf{Log Event:} Record deployment event with timestamp
    \item \textbf{Continue Telemetry:} Monitor descent
\end{enumerate}

\section{Appendix F: Safety Protocols}
\label{app:safety}

\subsection{Pre-Launch Safety Checklist}

\begin{enumerate}
    \item Battery fully charged and voltage >14.5V
    \item All igniters installed and continuity verified
    \item Telemetry link established with <5\% packet loss
    \item GPS lock acquired (>6 satellites)
    \item Parachutes properly packed with shock cord attached
    \item Explosion caps secured with O-rings installed
    \item All airframe connections tight and verified
    \item Launch site clear of personnel (>100 m radius)
    \item Ground station operational with clear data display
    \item Weather conditions acceptable (wind <10 m/s, no precipitation)
    \item Launch inhibit released only when ready for ignition
    \item Recovery team briefed and positioned
\end{enumerate}

\subsection{Launch Inhibit Conditions}

System will refuse to arm igniters if ANY of the following conditions exist:
\begin{itemize}
    \item Battery voltage <14.0V
    \item Igniter continuity not detected (drogue or main)
    \item Telemetry link not established
    \item GPS lock not acquired
    \item Barometer failure detected
    \item IMU failure detected
    \item Ground station command inhibit active
\end{itemize}

\subsection{Emergency Procedures}

\textbf{Lost Telemetry Link:}
\begin{itemize}
    \item System continues autonomous operation using onboard sensors
    \item Backup timer-based deployment activates if altitude data lost
    \item SD card logging continues for post-flight analysis
    \item GPS coordinates in last telemetry packet guide recovery
\end{itemize}

\textbf{Igniter Failure:}
\begin{itemize}
    \item Backup timer-based deployment triggers at T+[time] or altitude threshold
    \item Manual deployment via ground command (if telemetry available)
    \item Parachute extraction via motor ejection (if available on airframe)
\end{itemize}

\textbf{Premature Deployment:}
\begin{itemize}
    \item State machine guard prevents deployment below minimum altitude (100 m AGL)
    \item State machine guard prevents deployment before minimum time (10 seconds from liftoff)
    \item Dual-switch safety requires both software enable and hardware enable
\end{itemize}
